  <script>

    // ── 설정 ──────────────────────────────────────────

    const API_KEY = '371e7cad73f824fc2da187043a0d26d5';

    const GENRE_MAP = {

      28:'액션',18:'드라마',35:'코미디',27:'공포',

      10749:'로맨스',878:'SF',16:'애니',53:'스릴러',

      12:'어드벤처',14:'판타지',80:'범죄',99:'다큐'

    };

    const TIERS = ['S','A','B','C','D','F'];

    const MAX = 10;


    // ── 상태 ──────────────────────────────────────────

    let movies = {};      // { id: {id, title, poster, genres} }

    let placement = {};   // { id: 'S'|'A'|...|'unranked' }

    let dragId = null;


    // ── 초기 로드 ─────────────────────────────────────

    window.onload = () => {

      loadFromStorage();

      renderAll();

    };


    // ── 검색 ──────────────────────────────────────────

    async function searchMovies() {

      const query = document.getElementById('search-input').value.trim();

      if (!query) return;

      const url = `https://api.themoviedb.org/3/search/movie?api_key=${API_KEY}&query=${encodeURIComponent(query)}&language=ko-KR`;

      try {

        const res = await fetch(url);

        const data = await res.json();

        renderResults(data.results || []);

      } catch(e) {

        alert('검색 중 오류가 발생했습니다. API 키를 확인해주세요.');

      }

    }


    document.addEventListener('DOMContentLoaded', () => {

      document.getElementById('search-input')

        .addEventListener('keydown', e => { if(e.key==='Enter') searchMovies(); });

    });


    function renderResults(results) {

      const box = document.getElementById('search-results');

      box.innerHTML = '';

      results.slice(0, 10).forEach(m => {

        const div = document.createElement('div');

        div.className = 'result-card';

        div.onclick = () => addMovie(m);

        const imgHtml = m.poster_path

          ? `<img src="https://image.tmdb.org/t/p/w200${m.poster_path}" alt="${m.title}">`

          : `<div class="no-poster">포스터 없음</div>`;

        div.innerHTML = `${imgHtml}<div class="result-title">${m.title}</div>`;

        box.appendChild(div);

      });

    }


    // ── 영화 추가 ─────────────────────────────────────

    function addMovie(m) {

      const total = Object.keys(movies).length;

      if (total >= MAX) {

        showWarning();

        return;

      }

      const id = String(m.id);

      if (movies[id]) { alert('이미 추가된 영화입니다!'); return; }

      const genres = (m.genre_ids || [])

        .slice(0,2).map(g => GENRE_MAP[g] || '').filter(Boolean);

      movies[id] = { id, title: m.title, poster: m.poster_path || '', genres };

      placement[id] = 'unranked';

      saveToStorage();

      renderAll();

      document.getElementById('search-results').innerHTML = '';

      document.getElementById('search-input').value = '';

    }


    function showWarning() {

      const el = document.getElementById('warning-msg');

      el.style.display = 'block';

      setTimeout(() => { el.style.display = 'none'; }, 3000);

    }


    // ── 렌더링 ────────────────────────────────────────

    function renderAll() {

      document.getElementById('unranked').innerHTML = '';

      TIERS.forEach(t => { document.getElementById('tier-'+t).innerHTML = ''; });


      Object.values(movies).forEach(m => {

        const zone = placement[m.id] || 'unranked';

        const card = makeCard(m);

        if (zone === 'unranked') {

          document.getElementById('unranked').appendChild(card);

        } else {

          document.getElementById('tier-'+zone).appendChild(card);

        }

      });

    }


    function makeCard(m) {

      const div = document.createElement('div');

      div.className = 'movie-card';

      div.draggable = true;

      div.dataset.id = m.id;


      const imgHtml = m.poster

        ? `<img src="https://image.tmdb.org/t/p/w200${m.poster}" alt="${m.title}">`

        : `<div class="no-poster-card">${m.title}</div>`;


      const badges = m.genres.map(g => `<span class="badge">${g}</span>`).join('');


      div.innerHTML = `

        ${imgHtml}

        <div class="genre-badges">${badges}</div>

        <div class="card-title">${m.title}</div>

        <button class="delete-btn" onclick="deleteMovie('${m.id}')">✕</button>

      `;


      div.addEventListener('dragstart', e => {

        dragId = m.id;

        setTimeout(() => div.classList.add('dragging'), 0);

      });

      div.addEventListener('dragend', () => {

        div.classList.remove('dragging');

        dragId = null;

      });

      return div;

    }


    // ── 드롭 ──────────────────────────────────────────

    function onDrop(event, zone) {

      event.preventDefault();

      const dropEl = event.currentTarget;

      dropEl.classList.remove('drag-over');

      if (!dragId) return;

      placement[dragId] = zone;

      saveToStorage();

      renderAll();

    }


    // ── 삭제 ──────────────────────────────────────────

    function deleteMovie(id) {

      delete movies[id];

      delete placement[id];

      saveToStorage();

      renderAll();

    }


    // ── 초기화 ────────────────────────────────────────

    function resetAll() {

      if (!confirm('티어리스트를 초기화할까요?')) return;

      movies = {};

      placement = {};

      saveToStorage();

      renderAll();

    }


    // ── localStorage ──────────────────────────────────

    function saveToStorage() {

      localStorage.setItem('tier_movies', JSON.stringify(movies));

      localStorage.setItem('tier_placement', JSON.stringify(placement));

    }


    function loadFromStorage() {

      try {

        movies    = JSON.parse(localStorage.getItem('tier_movies'))    || {};

        placement = JSON.parse(localStorage.getItem('tier_placement')) || {};

      } catch(e) { movies = {}; placement = {}; }

    }


    function saveManual() {

      saveToStorage();

      const msg = document.getElementById('save-msg');

      msg.style.display = 'inline';

      setTimeout(() => { msg.style.display = 'none'; }, 2000);

    }


    // ── 카카오톡 공유 ─────────────────────────────────

    function shareKakao() {

      let text = '🎬 나의 영화 티어리스트\n\n';

      TIERS.forEach(t => {

        const ids = Object.keys(placement).filter(id => placement[id] === t);

        if (ids.length === 0) return;

        const titles = ids.map(id => movies[id]?.title || '').filter(Boolean).join(', ');

        text += `[${t}] ${titles}\n`;

      });

      const unrankedIds = Object.keys(placement).filter(id => placement[id] === 'unranked');

      if (unrankedIds.length > 0) {

        const titles = unrankedIds.map(id => movies[id]?.title || '').filter(Boolean).join(', ');

        text += `[미분류] ${titles}\n`;

      }

      const kakaoUrl = `https://story.kakao.com/share?url=${encodeURIComponent(location.href)}&text=${encodeURIComponent(text)}`;

      window.open(kakaoUrl, '_blank', 'width=600,height=400');

    }
